/** * Deals.gs — the quote deal / follow-up engine. * * Actions (routed in Router.gs): * quote.setDeal — change deal stage (Sent / Negotiating / Lost) * quote.win — payment-GATED: create Sales Order (+ items), mark quote Won * quote.followup — log a follow-up (+ reminder) with a next-action date * quote.revise — clone into a new revision (…-R{n}), supersede the prior * quote.delete — CEO-only: remove the quote + its line items * * Operates on the `quote` entity and the related `order` / `orderItem` / * `followup` / `reminder` entities (all generic Core CRUD — no schema surprises, * columns per docs/SCHEMAS.md & Bootstrap.gs CORE_HEADERS). */ /* ------------------------------------------------------------------ stage -- */ // quote.setDeal — Sent / Negotiating / Lost. (Won goes through quote.win.) function quoteSetDeal(payload) { const qno = payload && payload.quoteNo; const state = payload && payload.dealState; if (!qno || !state) throw new Error('quoteNo and dealState are required'); if (/won/i.test(state)) throw new Error('Use quote.win to mark Won — payment terms are required.'); const q = getRecord('quote', qno); if (!q) throw new Error('Quote not found: ' + qno); const ctx = _dealCtx(payload); const patch = { DealState: state, LastEditedBy: ctx.user, LastEditedAt: new Date() }; if (/lost/i.test(state)) { patch.LostReason = (payload && payload.lostReason) || ''; patch.DealClosedAt = new Date(); } updateRecord('quote', qno, patch, ctx); return { ok: true, quoteNo: qno, dealState: state }; } /* -------------------------------------------------------------------- win -- */ // quote.win — the payment gate. NO valid payment terms => NO Sales Order, ever. // payload: { quoteNo, paymentTerms:{type,advancePct,balanceDue,creditDays,ref}, // shipTo, notes, items:[{sku,name,qty,unit,rate,amount, // deliveryDate,source,vendor,inwardDate,poRef}] } function quoteWin(payload) { const qno = payload && payload.quoteNo; if (!qno) throw new Error('quoteNo is required'); const pt = (payload && payload.paymentTerms) || {}; if (!_validPaymentTerms(pt)) throw new Error('Payment terms are required to create the Sales Order.'); const q = getRecord('quote', qno); if (!q) throw new Error('Quote not found: ' + qno); if (String(q.SalesOrderNo || '')) throw new Error('Quote already converted to ' + q.SalesOrderNo); const ctx = _dealCtx(payload); const now = new Date(); const order = createRecord('order', { QuoteNo: qno, CustomerID: q.CustomerID || '', CustomerName: q.CustomerName || '', Phone: q.Phone || '', OrderDate: now, TotalAmount: q.GrandTotal || 0, PaymentType: pt.type || '', AdvancePct: pt.advancePct || '', BalanceDue: pt.balanceDue || '', CreditDays: pt.creditDays || '', PaymentRef: pt.ref || '', ShipTo: (payload && payload.shipTo) || '', Notes: (payload && payload.notes) || '', Status: 'Confirmed', CreatedBy: ctx.user, CreatedAt: now }, ctx); const orderNo = order.OrderID; const items = (payload && payload.items) || []; items.forEach(function (it) { createRecord('orderItem', { OrderID: orderNo, QuoteNo: qno, SKU: it.sku || '', Description: it.name || '', Qty: it.qty || 0, Unit: it.unit || '', Rate: it.rate || 0, Amount: it.amount || 0, ExpectedDelivery: it.deliveryDate || '', ExpectedDay: _dayName(it.deliveryDate), Source: it.source || 'stock', Vendor: it.vendor || '', ExpectedInward: it.inwardDate || '', InwardDay: _dayName(it.inwardDate), PORef: it.poRef || '' }, ctx); }); updateRecord('quote', qno, { DealState: 'Won', SalesOrderNo: orderNo, DealClosedAt: now, LastEditedBy: ctx.user, LastEditedAt: now }, ctx); return { ok: true, quoteNo: qno, orderNo: orderNo, items: items.length }; } // A payment term is valid only when the amounts are actually pinned down. function _validPaymentTerms(pt) { if (!pt || !pt.type) return false; const t = String(pt.type).toLowerCase(); if (t.indexOf('advance') >= 0 && t.indexOf('balance') >= 0) { const a = +pt.advancePct; return a > 0 && a <= 100; } if (t.indexOf('full') >= 0) return true; // full advance if (t.indexOf('credit') >= 0) return (+pt.creditDays) > 0; // credit N days if (t.indexOf('delivery') >= 0) return true; // pay on delivery return false; } /* --------------------------------------------------------------- followup -- */ // quote.followup — log a note + optional next-action date; raises a reminder. function quoteFollowup(payload) { const qno = payload && payload.quoteNo; const note = payload && payload.note; if (!qno || !note) throw new Error('quoteNo and note are required'); const q = getRecord('quote', qno); if (!q) throw new Error('Quote not found: ' + qno); const ctx = _dealCtx(payload); const now = new Date(); const next = (payload && payload.nextDate) || ''; createRecord('followup', { QuoteNo: qno, LeadID: q.LeadID || '', Customer: q.CustomerName || '', Phone: q.Phone || '', AssignedTo: ctx.user, DueAt: next, DealState: q.DealState || '', Note: note, CreatedAt: now, Status: next ? 'Open' : 'Done' }, ctx); if (next) { try { createRecord('reminder', { ForUser: (payload && payload.remindFor) || 'CEO', Type: 'followup', Message: 'Follow up: ' + (q.CustomerName || qno), DueAt: next, RefType: 'quote', RefID: qno, Status: 'Open', CreatedAt: now }, ctx); } catch (e) { /* reminder is best-effort */ } } return { ok: true, quoteNo: qno, nextDate: next }; } /* ---------------------------------------------------------------- revise -- */ // quote.revise — new revision keeps the base number with a -R{n} suffix, // supersedes the prior, and re-writes line items. Optionally takes edited lines. function quoteRevise(payload) { const qno = payload && payload.quoteNo; if (!qno) throw new Error('quoteNo is required'); const q = getRecord('quote', qno); if (!q) throw new Error('Quote not found: ' + qno); const ctx = _dealCtx(payload); const now = new Date(); const base = q.BaseQuoteNo || qno; const rev = (+q.Revision || 0) + 1; const newNo = String(base).replace(/-R\d+$/, '') + '-R' + rev; if (getRecord('quote', newNo)) throw new Error('Revision already exists: ' + newNo); const lines = (payload && payload.lines) || _safeJson(q.ItemsJSON); const withGst = (payload && payload.withGst !== undefined) ? payload.withGst : (String(q.WithGST) === 'true' || q.WithGST === true); const basic = _num(payload && payload.basicTotal, q.BasicTotal); const gst = _num(payload && payload.gstTotal, q.GSTTotal); const grand = _num(payload && payload.grandTotal, q.GrandTotal); const carryState = (q.DealState && !/lost|won/i.test(q.DealState)) ? q.DealState : 'Sent'; createRecord('quote', { QuoteNo: newNo, BaseQuoteNo: base, Revision: rev, LeadID: q.LeadID || '', CustomerID: q.CustomerID || '', CustomerName: (payload && payload.customerName) || q.CustomerName || '', Phone: (payload && payload.phone) || q.Phone || '', OutputMode: (payload && payload.outputMode) || q.OutputMode || 'proposal', WithGST: withGst, BasicTotal: basic, GSTTotal: gst, GrandTotal: grand, Status: 'Sent', CreatedBy: ctx.user, CreatedAt: now, LastEditedBy: ctx.user, LastEditedAt: now, LastSentAt: now, ItemsJSON: JSON.stringify(lines), DealState: carryState }, ctx); (lines || []).forEach(function (l) { const rate = +l.rate || 0, qty = +l.qty || 0, g = +l.gst || 0, amt = rate * qty; createRecord('quoteItem', { QuoteNo: newNo, SKU: l.sku || '', Description: l.name || '', Qty: qty, Unit: l.unit || '', Rate: rate, Amount: amt, GSTPct: g, GSTAmount: amt * (g / 100), LineTotal: amt * (1 + g / 100) }, ctx); }); updateRecord('quote', qno, { Status: 'Superseded', LastEditedBy: ctx.user, LastEditedAt: now }, ctx); return { ok: true, quoteNo: newNo, revision: rev, supersededFrom: qno }; } /* ---------------------------------------------------------------- delete -- */ // quote.delete — CEO only. Removes the quote header + all its line items. function quoteDelete(payload, context) { const qno = payload && payload.quoteNo; if (!qno) throw new Error('quoteNo is required'); const role = String((context && context.role) || (payload && payload.role) || '').toLowerCase(); const acc = String((payload && payload.accessLevel) || '').toLowerCase(); if (!/ceo|founder/.test(role) && !/ceo|founder/.test(acc)) { throw new Error('Only the CEO can delete a quote.'); } const ctx = _dealCtx(payload); const items = listRecords('quoteItem', { QuoteNo: qno }) || []; items.forEach(function (it) { try { deleteRecord('quoteItem', it.QuoteItemID, ctx); } catch (e) {} }); deleteRecord('quote', qno, ctx); return { ok: true, quoteNo: qno, itemsDeleted: items.length }; } /* ------------------------------------------------------------ deploy help -- */ // Run ONCE after deploying the deal engine. For every entity the engine writes // to, this creates its tab if missing and ADDS any columns the schema has but // the live tab lacks (bootstrapCore only creates missing tabs, it does not // migrate headers — and the Quotes tab predates DealState/SalesOrderNo/etc.). // Idempotent — safe to re-run. function ensureDealTabs() { const entities = ['quote', 'order', 'orderItem', 'followup', 'reminder']; const log = []; entities.forEach(function (e) { const cfg = ENTITIES[e], headers = CORE_HEADERS[e]; if (!cfg || !headers) { log.push('skip ' + e + ' (no config/headers)'); return; } const ss = SpreadsheetApp.openById(extractSpreadsheetId(SPREADSHEETS[cfg.spreadsheet])); let sh = ss.getSheetByName(cfg.tab); if (!sh) { sh = ss.insertSheet(cfg.tab); sh.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold'); sh.setFrozenRows(1); log.push('created ' + cfg.tab + ' (' + headers.length + ' cols)'); return; } const have = sh.getRange(1, 1, 1, Math.max(1, sh.getLastColumn())).getValues()[0].map(String); const missing = headers.filter(function (h) { return have.indexOf(h) < 0; }); if (missing.length) { sh.getRange(1, have.length + 1, 1, missing.length).setValues([missing]).setFontWeight('bold'); log.push('added to ' + cfg.tab + ': ' + missing.join(', ')); } else { log.push(cfg.tab + ' already up to date'); } }); const out = log.join('\n'); Logger.log(out); return out; } /* ---------------------------------------------------------------- utils -- */ function _dealCtx(payload) { return { user: (payload && payload.by) || (payload && payload.user) || 'app', role: 'app', device: 'web' }; } function _num(v, fallback) { return (v !== undefined && v !== null && v !== '') ? +v : (+fallback || 0); } function _safeJson(s) { try { return JSON.parse(s || '[]'); } catch (e) { return []; } } function _dayName(d) { if (!d) return ''; const dt = (d instanceof Date) ? d : new Date(d); if (isNaN(dt.getTime())) return ''; return ['Sunday','Monday','Tuesday','Wednesday','Thursday','Friday','Saturday'][dt.getDay()]; }
